    const straight = edges.filter(e => !e.self);
    const step = cardW + gap;
    natOrder.forEach((t, i) => { coords.get(t).col = i; });

    // le corsie sono gli spazi tra le colonne: la corsia g sta a sinistra della colonna g (0 = margine sinistro)
    const gapRange = g => { const a = g === 0 ? sideMargin - gap : sideMargin + (g - 1) * step + cardW; return [a, a + gap]; };

    straight.forEach(e => {
      if (e.src.col !== undefined) {                    // tabella nativa che punta a un'altra nativa
        const sc = e.src.col, dc = e.dst.col;
        e.gapD = sc > dc ? dc + 1 : dc;                 // corsia della destinazione rivolta verso la sorgente
        if (Math.abs(sc - dc) > 1) {                    // colonne non adiacenti: si passa sopra le schede
          e.corridor = true;
          e.gapS = sc > dc ? sc : sc + 1;               // corsia della sorgente rivolta verso la destinazione
        }
      } else {                                          // tabella ponte: corsia dal lato rivolto verso di lei
        const right = e.dst.col + 1, left = e.dst.col;  // (o dall'altro se è coperta dal ponte stesso)
        const covered = g => { const r = gapRange(g), m = (r[0] + r[1]) / 2; return m > e.src.x - 4 && m < e.src.x + e.src.w + 4; };
        let g = e.src.x > e.dst.x ? right : left;
        if (covered(g)) g = g === right ? left : right;
        e.gapD = g;
      }
    });

    // ogni linea ha la sua x nella corsia, anche se la corsia è condivisa
    const lanes = new Map();
    const inLane = (g, it) => { if (!lanes.has(g)) lanes.set(g, []); lanes.get(g).push(it); };
    straight.forEach(e => {
      inLane(e.gapD, { e: e, key: 'laneX' });
      if (e.corridor) inLane(e.gapS, { e: e, key: 'laneSX' });
    });
    lanes.forEach((grp, g) => {
      const r = gapRange(g), spacing = (r[1] - r[0] - 16) / grp.length;
      grp.forEach((it, i) => { it.e[it.key] = r[0] + 8 + spacing * (i + 0.5); });
    });
    let corr = 0;
    straight.forEach(e => { if (e.corridor) e.corrY = topY - 20 - 14 * corr++; });   // corridoi sopra le schede

    let lines = '';
    const labels = [];
    const arrow = (tipX, y, dir) => {
      const base = tipX - dir * 8;
      grow(Math.min(tipX, base), y - 4, Math.max(tipX, base), y + 4);
      return '<polygon points="' + tipX + ',' + y + ' ' + base + ',' + (y - 4) + ' ' + base + ',' + (y + 4) + '" fill="#2458a6"/>';
    };
    const pathSvg = pts => {
      pts.forEach(p => grow(p[0], p[1], p[0], p[1]));
      return '<path d="M ' + pts.map(p => p[0] + ' ' + p[1]).join(' L ') + '" fill="none" stroke="#2458a6" stroke-width="1.8" stroke-linejoin="round"/>';
    };

    straight.forEach(e => {
      const enterX = e.laneX < e.dst.x + e.dst.w / 2 ? e.dst.x : e.dst.x + e.dst.w;
      let pts;
      if (e.corridor) {
        const exitX = e.laneSX < e.src.x + e.src.w / 2 ? e.src.x : e.src.x + e.src.w;

        pts = [[exitX, e.srcY], [e.laneX, e.srcY], [e.laneX, e.dstY], [enterX, e.dstY]];

      }
      lines += pathSvg(pts);
      lines += arrow(enterX > e.laneX ? enterX - 1 : enterX + 1, e.dstY, enterX > e.laneX ? 1 : -1);
    });

    const placed = [];
    let labelSvg = '';
    labels.forEach(lb => {
      const w = lb.text.length * 6.4 + 14, h = 18;
      const vertical = lb.axis !== 'h';
      const hit = (x, y, o) => Math.abs(x - o.cx) < (w + o.w) / 2 + 2 && Math.abs(y - o.cy) < h + 2;
      const shifts = vertical ? [0, 22, -22, 44, -44, 66, -66] : [0, w + 10, -(w + 10), 2 * (w + 10), -2 * (w + 10)];
      const lo = vertical ? lb.min : lb.min + w / 2, hi = vertical ? lb.max : lb.max - w / 2;
      let cx = lb.cx, cy = lb.cy;
      for (let i = 0; i < shifts.length; i++) {
        const nx = vertical ? lb.cx : lb.cx + shifts[i], ny = vertical ? lb.cy + shifts[i] : lb.cy;
        const v = vertical ? ny : nx;
        if ((hi < lo || (v >= lo && v <= hi)) && !placed.some(o => hit(nx, ny, o))) { cx = nx; cy = ny; break; }

8. Funzione relationalHtml
======================================================================
DOVE: come nuova sezione, subito prima della sezione 'STATO E INTERFACCIA'

  // ============================================================
  //  MODELLO RELAZIONALE (forma testuale)
  // ============================================================
  // NOME (chiave primaria sottolineata, chiave esterna in corsivo, altri campi): senza tipi.
  // Si usano i tag <u> e <i> perché sopravvivono al copia e incolla in Word o LibreOffice.
  function relationalHtml(tables) {
    const col = c => {
      const n = esc(c.name);
      if (c.pk && c.fk) return '<u><i>' + n + '</i></u>';
      if (c.pk) return '<u>' + n + '</u>';
      if (c.fk) return '<i>' + n + '</i>';
      return n;
    };
    let html = '';
    tables.forEach(t => {
      html += '<p class="rel"><b>' + esc(t.name.toUpperCase()) + '</b> (' + orderedCols(t).map(col).join(', ') + ')</p>';
    });
    const refs = [];
    const list = a => a.length > 1 ? '(' + a.join(', ') + ')' : a[0];
    tables.forEach(t => t.fks.forEach(f => {
      refs.push('<p class="ref">' + esc(t.name.toUpperCase()) + '.' + esc(list(f.cols)) + ' &rarr; ' +
                esc(f.ref.name.toUpperCase()) + '.' + esc(list(f.refCols)) + '</p>');
    }));
    if (refs.length) html += '<p class="refhead">Vincoli di integrit\u00e0 referenziale</p>' + refs.join('');
    return html;
  }

  // messaggio breve sul pulsante dopo la copia
  function flash(btn, ok) {
    const old = btn.textContent;
    btn.textContent = ok ? 'Copiato' : 'Copia non riuscita';
    setTimeout(() => { btn.textContent = old; }, 1800);
  }

  $('btnCopy').addEventListener('click', () => {
    if (!current) return;
    const btn = $('btnCopy');
    const fallback = () => {
      const ta = document.createElement('textarea');
      ta.value = current.sql;
      document.body.appendChild(ta);
      ta.select();
      let ok = false;
      try { ok = document.execCommand('copy'); } catch (err) { ok = false; }
      document.body.removeChild(ta);
      flash(btn, ok);
    };
    if (navigator.clipboard && navigator.clipboard.writeText) {
      navigator.clipboard.writeText(current.sql).then(() => flash(btn, true), fallback);
    } else {
      fallback();
    }
  });

  // copia il modello con la formattazione (sottolineato e corsivo), pronto da incollare in un documento
  $('btnCopyModel').addEventListener('click', () => {
    if (!current) return;
    const sel = window.getSelection(), range = document.createRange();
    range.selectNodeContents($('modelBody'));
    sel.removeAllRanges();
    sel.addRange(range);
    let ok = false;
    try { ok = document.execCommand('copy'); } catch (err) { ok = false; }
    sel.removeAllRanges();
    flash($('btnCopyModel'), ok);
  });
