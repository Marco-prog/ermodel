9. Sezione tocco
======================================================================
DOVE: come nuova sezione, subito prima della sezione AVVIO

  // ============================================================
  //  TOCCO (tablet): pizzico a due dita, doppio tocco, rinomina al rilascio
  // ============================================================
  // I gestori di tela, elementi e attributi usano i Pointer Events: lo stesso codice vale per mouse, dito e penna.
  // Qui c'è ciò che serve solo al tocco.
  const touches = new Map();                 // dita appoggiate: pointerId -> { x, y }
  let pinch = null;                          // gesto a due dita in corso
  let downCount = 0;                         // puntatori attualmente premuti
  let pendingRename = null;                  // rinomina da aprire al rilascio (la tastiera si apre solo dopo un gesto concluso)
  let tap = null, lastTap = null, linkTap = null;

  function touchThr(e) { return (e.originalEvent || e).pointerType === 'mouse' ? 0 : 8; }   // soglia di movimento prima di trascinare

  document.addEventListener('pointerdown', function (e) {
    if (e.isPrimary) downCount = 0;          // un rilascio perso non deve bloccare il conteggio
    downCount++;
    if (e.pointerType !== 'touch') { tap = null; return; }
    touches.set(e.pointerId, { x: e.clientX, y: e.clientY });
    if (touches.size === 2) {                // seconda dita: inizia il pizzico, il resto viene annullato
      drag = null; pan = null; linkTap = null; pendingRename = null; tap = null; lastTap = null;
      $svg.removeClass('panning');
      const a = Array.from(touches.values()), r = svg.getBoundingClientRect();
      const mx = (a[0].x + a[1].x) / 2 - r.left, my = (a[0].y + a[1].y) / 2 - r.top;
      pinch = { d: Math.hypot(a[0].x - a[1].x, a[0].y - a[1].y) || 1, k: view.k, px: view.x + mx / view.k, py: view.y + my / view.k };
    }
    if (touches.size >= 2) { e.stopPropagation(); return; }
    const el = e.target.closest ? e.target.closest('.item, .attr') : null;      // per riconoscere il doppio tocco
    tap = el ? { kind: el.classList.contains('attr') ? 'attribute' : el.getAttribute('data-kind'),
                 id: el.getAttribute('data-id'), x: e.clientX, y: e.clientY } : null;
  }, true);

  document.addEventListener('pointermove', function (e) {
    if (e.pointerType !== 'touch' || !touches.has(e.pointerId)) return;
    touches.set(e.pointerId, { x: e.clientX, y: e.clientY });
    if (!pinch || touches.size !== 2) return;
    const a = Array.from(touches.values()), r = svg.getBoundingClientRect();
    const mx = (a[0].x + a[1].x) / 2 - r.left, my = (a[0].y + a[1].y) / 2 - r.top;
    const k = Math.min(ZOOM_MAX, Math.max(ZOOM_MIN, pinch.k * Math.hypot(a[0].x - a[1].x, a[0].y - a[1].y) / pinch.d));
    view.k = k;                              // zoom e spostamento con le due dita, il punto di mezzo resta fermo
    view.x = pinch.px - mx / k;
    view.y = pinch.py - my / k;
    applyView();
  }, true);

  function pointerEnd(e, cancelled) {
    downCount = Math.max(0, downCount - 1);
    if (e.pointerType === 'touch') {
      touches.delete(e.pointerId);
      if (touches.size < 2) pinch = null;
    }
    if (cancelled) {
      drag = null; pan = null; linkTap = null; tap = null; pendingRename = null;
      $svg.removeClass('panning');
      return;
    }
    if (linkTap) {                           // collegamento: clic o tocco senza spostamento alterna 1 / molti
      const t = linkTap;
      linkTap = null;
      if (Math.hypot(e.clientX - t.sx, e.clientY - t.sy) < 8) {
        const l = model.links.find(x => x.id === t.id);
        if (l) { l.many = !l.many; drawLinks(); pushHistory(); }
      }
    }
    if (e.pointerType === 'touch' && tap && addMode === null && Math.hypot(e.clientX - tap.x, e.clientY - tap.y) < 10) {
      const now = Date.now();                // doppio tocco sullo stesso elemento: rinomina
      if (lastTap && lastTap.id === tap.id && lastTap.kind === tap.kind && now - lastTap.t < 350) {
        lastTap = null;
        startRename(tap.kind, tap.id);
      } else {
        lastTap = { kind: tap.kind, id: tap.id, t: now };
      }
    }
    tap = null;
    if (downCount === 0 && pendingRename) {  // rinomina rimandata dal momento della pressione
      const p = pendingRename;
      pendingRename = null;
      startRename(p.kind, p.id);
    }
  }
  document.addEventListener('pointerup', e => pointerEnd(e, false));
  document.addEventListener('pointercancel', e => pointerEnd(e, true));
  svg.addEventListener('contextmenu', e => e.preventDefault());       // la pressione lunga non apre menu
