5. addAssocAttrs
======================================================================
DOVE: SOSTITUISCI l'intera funzione addAssocAttrs con

    // Attributi dell'associazione. In una tabella di associazione (ownTable) quelli che sono identificatori formano
    // la chiave primaria propria (id diventa id_<associazione>). Se l'associazione non genera una tabella (1:N, 1:1)
    // i suoi attributi finiscono in quella di un'entità, che ha già la sua chiave: sono colonne normali.
    function addAssocAttrs(t, a, ownTable) {
      a.attrs.forEach(at => {
        if (at.pk && ownTable) {
          const pure = words(at.name).join('') === 'id';
          addCol(t, pure ? 'id_' + t.name : at.name, { assoc: true, pk: true, type: isIdName(at.name) ? 'INT' : guessType(at.name) });
        } else {
          if (at.pk) warn('L\u2019identificatore «' + at.name + '» dell\u2019associazione «' + a.name + '» è una colonna normale: l\u2019associazione non genera una tabella propria.');
          addCol(t, at.name, { assoc: true, type: guessType(at.name) });
        }
      });
    }
